const app = express();
app.use(express.urlencoded({ extended: true }));
app.use(express.json());
app.use('/ivr', ivrRouter);

function extractSid(xml) {
  const m = String(xml).match(/sid=([a-f0-9-]+)/);
  if (!m) throw new Error('sid not found');
  return m[1];
}

async function newCall(callSid) {
  const res = await request(app).post('/ivr/voice').send({ CallSid: callSid || 'CA-test' });
  expect(res.status).toBe(200);
  return extractSid(res.text);
}

    expect(session.stage).toBe('court');
    expect(res.text).toContain('/ivr/court');
